Schema::create('notifications', function (Blueprint $table) {
    $table->id();

    // penerima notif
    $table->foreignId('user_id')->constrained()->cascadeOnDelete();

    // jenis notif
    $table->string('type');

    // judul notif
    $table->string('title');

    // isi notif
    $table->text('message');

    // status baca
    $table->timestamp('read_at')->nullable();

    // icon opsional
    $table->string('icon')->nullable();

    // warna notif
    $table->string('color')->nullable();

    $table->timestamps();
});
